cmake_minimum_required(VERSION 3.16)
project(controller_ui_acceptance C)
set(CMAKE_C_STANDARD 11)
if(NOT LVGL_SOURCE_DIR)
  message(FATAL_ERROR "Set LVGL_SOURCE_DIR to an LVGL v8.3.11 checkout")
endif()
file(GLOB_RECURSE LVGL_SOURCES CONFIGURE_DEPENDS "${LVGL_SOURCE_DIR}/src/*.c")
add_library(lvgl STATIC ${LVGL_SOURCES})
target_compile_definitions(lvgl PUBLIC LV_CONF_INCLUDE_SIMPLE=1)
target_include_directories(lvgl PUBLIC "${LVGL_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(ui_acceptance acceptance.c ../../firmware/esp32/main/controller_ui.c)
target_include_directories(ui_acceptance PRIVATE ../../firmware/esp32/main)
target_link_libraries(ui_acceptance PRIVATE lvgl m)
target_compile_options(ui_acceptance PRIVATE -Wall -Wextra -Werror)
